from pathlib import Path
from datetime import datetime, timezone
import hashlib, json, math
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

repo=Path('/workspace/Safety-Dial')
source=repo/'docs/mainPlan/results/continuation-v4/decomposition/decomposition.json'
audit=repo/'docs/mainPlan/results/continuation-v4/development_diagnosis_audit.json'
out=repo/'runs/continuation-audit/report-figures-20260927'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
inputs={str(p):sha(p) for p in (source,audit,Path(__file__))}
report=json.loads(source.read_text())
verified=json.loads(audit.read_text())
dev=report['banks']['dev']; decision=dev['at_m0']; attribution=decision['attribution']; imagined=decision['imagined']; gate=report['gate']
assert {key:attribution[key] for key in ('temporal','readout','imagination')} == verified['independent_attribution_counts']
assert gate==verified['gate']
assert sum(attribution[key] for key in ('temporal','readout','imagination'))==gate['n_attributable_false_safe']==28
assert attribution['imagination']==gate['n_imagination']==21 and attribution['readout']==7 and attribution['temporal']==0
assert imagined['n_accepted_censored']==1 and not math.isfinite(imagined['fsa'])
assert imagined['n_accepted']==163 and dev['n_rows']==192 and dev['n_roots']==24
assert decision['m']==0 and gate['role']=='development' and gate['passes'] is True
out.mkdir(exist_ok=True)
with plt.rc_context({'font.family':'DejaVu Sans','font.size':12,'pdf.fonttype':42}):
 fig=plt.figure(figsize=(7.0,4.8),facecolor='white')
 fig.text(.055,.96,'E1: where known development false-safes arose',fontsize=15,weight='bold',va='top',color='#172536')
 fig.text(.055,.895,'Attribution denominator: 28 known false-safe decisions',fontsize=12,color='#33465b')
 fig.text(.055,.845,'Margin 0 px; 192 decisions from 24 development roots',fontsize=10.5,color='#526176')
 ax=fig.add_axes([.31,.40,.62,.34])
 names=['Imagined dynamics','Real-image readout','Temporal sampling'];values=[21,7,0]
 ax.barh([2,1,0],values,height=.6,color=['#315f80','#3a8b88','#969ea8'])
 ax.set_yticks([2,1,0],names,fontsize=11.5)
 ax.set_xlim(0,28);ax.set_xticks([0,7,14,21,28]);ax.tick_params(axis='y',length=0,pad=10)
 ax.tick_params(axis='x',labelsize=10.5,colors='#526176')
 ax.set_xlabel('Attributed decision count',fontsize=11,labelpad=9)
 ax.grid(axis='x',color='#dbe1e8',linewidth=.7);ax.set_axisbelow(True)
 for spine in ('top','right','left'):ax.spines[spine].set_visible(False)
 ax.spines['bottom'].set_color('#a3afbc')
 for y,value in zip([2,1,0],values):ax.text(value+.45,y,str(value),va='center',fontsize=12,weight='bold',color='#172536')
 fig.text(.055,.22,'Recorded gate: passes   |   Imagination share: 75%',fontsize=11.5,weight='bold',color='#172536')
 fig.text(.055,.145,'1 accepted unresolved future is separate from these 28.',fontsize=11,color='#804300')
 fig.text(.055,.09,'Full-data FSA is undefined. Domain exits and censored futures\nare excluded from attribution; these counts are not an effect estimate.',fontsize=10.2,va='top',color='#33465b')
 fig.savefig(out/'e1-development-attribution.png',dpi=220)
 fig.savefig(out/'e1-development-attribution.pdf')
 plt.close(fig)
