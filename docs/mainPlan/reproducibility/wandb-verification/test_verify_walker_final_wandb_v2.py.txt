def test_complete_exact_history_one_window_and_no_remote_write(fixture):
    f = fixture
    f.mod.main()
    result = json.loads(f.output.read_text())
    assert result["recovered_training_history_records"] == 2139
    assert result["full_saved_report_history_records"] == 10739
    assert result["validation_steps"] == list(range(176000, 212001, 4000)) + [214780]
    assert result["remote_write_calls"] == 0
    assert f.calls[-1] == (
        "scan_history",
        {"page_size": 1_000_000, "min_step": 0, "max_step": 214781, "use_cache": False},
    )
    assert f.calls[0][1] == {"timeout": 60, "host": "https://api.wandb.ai", "explicit_auth": True}


@pytest.mark.parametrize(
    "change", ["running", "returncode", "wrong_config", "config_bytes", "report_hash"]
)
def test_local_barrier_stops_before_api(fixture, change):
    f = fixture
    if change == "running":
        f.state["stages"]["lewm_a_full"]["status"] = "running"
    elif change == "returncode":
        f.state["stages"]["lewm_a_full"]["attempts"][0]["returncode"] = 2
    elif change == "wrong_config":
        f.state["config_sha256"] = "0" * 64
    elif change == "config_bytes":
        f.config_path.write_text("changed recipe")
    else:
        f.rp.write_text(f.rp.read_text() + " ")
    f.sp.write_text(json.dumps(f.state))
    with pytest.raises(ValueError):
        f.mod.main()
    assert not f.calls and not f.output.exists()

@pytest.mark.parametrize(
    "change",
    [
        "missing",
        "duplicate",
        "before_resume",
        "fractional_step",
        "metric",
        "nan",
        "missing_validation",
    ],
)
def test_history_and_censorless_step_identity_are_exact(fixture, change):
    f = fixture
    if change == "missing":
        f.remote_rows.pop(0)
    elif change == "duplicate":
        f.remote_rows.insert(1, dict(f.remote_rows[0]))
    elif change == "before_resume":
        f.remote_rows[0].update(step=172000, _step=172000)
    elif change == "fractional_step":
        f.remote_rows[0]["step"] += 0.5
    elif change == "metric":
        f.remote_rows[0]["pred_loss"] += 0.01
    elif change == "nan":
        f.remote_rows[0]["lr"] = float("nan")
    else:
        next(v for v in f.remote_rows if "val_pred_loss" in v).pop("val_pred_loss")
    with pytest.raises(ValueError):
        f.mod.main()
    assert not f.output.exists()

def test_hard_deadline_exits_even_during_blocking_sleep():
    code = f"""import importlib.util,signal,time
s=importlib.util.spec_from_file_location("v", {str(SOURCE)!r})
m=importlib.util.module_from_spec(s);s.loader.exec_module(m)
signal.signal(signal.SIGALRM,m.hard_deadline)
signal.setitimer(signal.ITIMER_REAL,0.1)
time.sleep(10)
"""
    start = time.monotonic()
    proc = subprocess.run([sys.executable, "-c", code], capture_output=True, text=True, timeout=3)
    assert proc.returncode == 124 and time.monotonic() - start < 3
    assert json.loads(proc.stdout)["reason"] == "overall_process_deadline"
    assert "Traceback" not in proc.stderr
