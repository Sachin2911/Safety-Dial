def load():
    spec = importlib.util.spec_from_file_location('s3_remote', SOURCE)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def write(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value))

def test_negative_health_gate_is_valid(built):
    m, p, a, audit, state, refresh = built
    for key in ('run', 'results'):
        report = json.loads((p[key] / 'gate.json').read_text())
        report['gate'] = {'go': False, 'active_rules': []}
        write(p[key] / 'gate.json', report)
    stage = state['stages']['physical_probe_gate']
    stage['status'] = state['status'] = 'gate_stopped'
    stage['attempts'][-1]['returncode'] = 2
    state['stages']['prospective_acquisition'] = {'status': 'pending', 'attempts': []}
    audit['s3']['gate'] = report['gate']
    audit['s3']['report_sha256'] = m.file_record(p['run'] / 'gate.json')['sha256']
    write(a, audit)
    refresh()
    assert m.prepare(a)['gate_go'] is False

    write(p['state'], state)
    with pytest.raises(m.AuditError, match='unattempted'):
        m.prepare(a)

class FakeAPI:
    def __init__(self, m, receipt, files):
        self.m, self.receipt, self.files = m, receipt, files
        self.info = SimpleNamespace(id=m.REPO_ID, private=True, sha=receipt['revision'])
        self.tag = SimpleNamespace(name=m.RUN_ID, target_commit=receipt['revision'])
        self.entries = []
        self.calls = []
        self.refs_calls = 0
        for name, rec in files.items():
            if Path(name).name in m.IGNORED:
                continue
            lfs = SimpleNamespace(sha256=rec['sha256'], size=rec['size_bytes']) if name.endswith('.pt') else None
            self.entries.append(SimpleNamespace(path=f"{receipt['path']}/{name}", size=rec['size_bytes'],
                                                lfs=lfs, blob_id=rec['git_blob_sha1']))
        self.change_tag_after = False
    def repo_info(self, **kw):
        self.calls.append(('repo_info', kw))
        assert kw['revision'] == self.receipt['revision']
        return self.info
    def list_repo_refs(self, **kw):
        self.calls.append(('list_repo_refs', kw))
        self.refs_calls += 1
        tag = copy.deepcopy(self.tag)
        if self.change_tag_after and self.refs_calls > 1:
            tag.target_commit = 'd' * 40
        return SimpleNamespace(tags=[tag])
    def list_repo_tree(self, **kw):
        self.calls.append(('list_repo_tree', kw))
        assert kw['revision'] == self.receipt['revision']
        return iter(self.entries)


def fake_remote(built):
    m, _, audit, _, _, _ = built
    p = m.prepare(audit)
    return m, p, FakeAPI(m, p['receipt'], p['trees']['run'])


def test_metadata_checks_all_payload_and_receipt_exclusion(built):
    m, p, api = fake_remote(built)
    result = m.compare_remote(api, p['receipt'], p['trees']['run'])
    assert len(result) == 9 and 'hf_upload.json' not in result
    assert result['walker_mlp.pt']['remote_match_method'] == 'LFS SHA256 and size'
    assert result['gate.json']['remote_match_method'] == 'canonical Git blob SHA1 and size'
    assert api.refs_calls == 2
    assert set(name for name, _ in api.calls) == {'repo_info', 'list_repo_refs', 'list_repo_tree'}


@pytest.mark.parametrize('mutation', ['private','revision','tag','missing','extra','lfs','git','size','duplicate','tag_after','receipt_uploaded'])
def test_remote_failures_refused(built, mutation):
    m, p, api = fake_remote(built)
    if mutation == 'private':api.info.private = False
    elif mutation == 'revision':api.info.sha = 'd' * 40
    elif mutation == 'tag':api.tag.target_commit = 'd' * 40
    elif mutation == 'missing':api.entries.pop()
    elif mutation == 'extra':api.entries.append(SimpleNamespace(path=p['receipt']['path']+'/extra',size=1,lfs=None,blob_id='x'))
    elif mutation == 'lfs':next(x for x in api.entries if x.lfs).lfs.sha256 = 'bad'
    elif mutation == 'git':next(x for x in api.entries if not x.lfs).blob_id = 'bad'
    elif mutation == 'size':api.entries[0].size += 1
    elif mutation == 'duplicate':api.entries.append(api.entries[0])
    elif mutation == 'tag_after':api.change_tag_after = True
    elif mutation == 'receipt_uploaded':api.entries.append(SimpleNamespace(path=p['receipt']['path']+'/hf_upload.json',size=1,lfs=None,blob_id='x'))
    with pytest.raises(m.AuditError):m.compare_remote(api,p['receipt'],p['trees']['run'])
