import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path
import numpy as np

root=Path('/workspace/Safety-Dial')
snapshot=root/'runs/maniguard/reports/jar-repeat-progress-0028.json'
data=json.loads(snapshot.read_text())
indices=[17,14,10,26]
records=[next(r for r in data['completed_diagnostics'] if r['index']==i) for i in indices]
paths=[Path(r['diagnostic']['output']) for r in records]

def read(path): return json.loads(path.read_text())
def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()

jobs=[read(p/'job.json') for p in paths]
configs=[read(p/'episode/eval_config.json') for p in paths]
runtimes=[read(p/'runtime-provenance.json') for p in paths]
for job in jobs[1:]:
    for key in ['family','scene','seed','policy','code_hashes']:
        assert job[key]==jobs[0][key]
for config in configs[1:]:
    assert {k:v for k,v in config.items() if k!='output_dir'} == {
        k:v for k,v in configs[0].items() if k!='output_dir'}
assert all(r==runtimes[0] for r in runtimes)
for path,job in zip(paths,jobs):
    for name,expected in job['code_hashes'].items():
        assert sha(path/'study_source'/name)==expected
    assert sha(path/'benchmark_instrumented.py')==runtimes[0]['runtime_sha256']
    record=next(r for r in records if Path(r['diagnostic']['output'])==path)
    assert sha(path/'trace.jsonl')==record['trace_sha256']
control,candidate=paths[0],paths[-1]
arrays=[]
with np.load(candidate/'initial_observation.npz') as ca,np.load(control/'initial_observation.npz') as co:
    assert ca.files==co.files
    for name in ca.files:
        a,b=ca[name],co[name]
        assert a.shape==b.shape and a.dtype==b.dtype
        arrays.append({'key':name,'shape':list(a.shape),'dtype':str(a.dtype),
            'exactly_equal':bool(np.array_equal(a,b)),
            'max_absolute_difference':float(np.max(np.abs(a.astype(float)-b.astype(float)))),
            'candidate_sha256':hashlib.sha256(a.tobytes()).hexdigest(),
            'control_sha256':hashlib.sha256(b.tobytes()).hexdigest()})
first=[];metadata=[]
for path in [candidate,control]:
    for line in (path/'trace.jsonl').open():
        row=json.loads(line)
        if row['type']=='policy_metadata':metadata.append(row['metadata'])
        if row['type']=='proposal':
            first.append(np.asarray(row['original']))
            break
assert metadata[0]==metadata[1]
motion=records[-1]
assert motion['diagnostic']['safe_completion']
assert motion['smoothed_target_cap_activation']['actions']==0
assert motion['gripper_commands_unchanged']
assert motion['diagnostic']['adapter']['settle_steps']==0
assert all(r['timeline']['first_target_held_observation'] is None for r in records[:-1])
source=root/'runs/maniguard/reports/jar-repeat-first-smoothing-gain-audit.py'
assert not source.exists()
source.write_bytes(Path(__file__).read_bytes())
result={'recorded_at_utc':datetime.now(timezone.utc).strftime('%Y-%m-%dT%H:%M:%SZ'),
    'scope':'All four completed conditions on Jar task_0013, root one, within the unfinished 48-trial engineering comparison. Separate physical starts, not deterministic counterfactuals.',
    'case_conditions':records,
    'shared_policy':jobs[0]['policy'],'shared_code_hashes':jobs[0]['code_hashes'],
    'shared_runtime_provenance':runtimes[0],'different_evaluation_fields':['output_dir'],
    'smoothing_identity_input_comparison':{'candidate_episode':str(candidate.relative_to(root)),
        'control_episode':str(control.relative_to(root)),'initial_arrays':arrays,
        'first_original_proposal_equal':bool(np.array_equal(first[0],first[1])),
        'first_original_proposal_max_absolute_difference':float(np.max(np.abs(first[0]-first[1])))},
    'evidence_hashes':{str((path/name).relative_to(root)):sha(path/name) for path in paths
        for name in ['job.json','trace.jsonl','initial_observation.npz','episode/eval_config.json']},
    'source_progress_snapshot':{'path':str(snapshot.relative_to(root)),'sha256':sha(snapshot)},
    'analysis_source':{'path':str(source.relative_to(root)),'sha256':sha(source)},
    'interpretation':'Smoothing completes safely in 1082 actions and 136 queries; identity and handback remain unsafe unfinished, while the plain guard remains safe unfinished. Caps do not activate and gripper commands are unchanged. Obtaining a target grasp differs as well as safety. Initial images and first policy proposals differ before the first adapter action, so this does not isolate a causal repair of abrupt movement. The guard variants use privileged simulator information; smoothing does not. Remaining scenes and controls are needed before judging aggregate benefit.'}
output=root/'docs/research/research_notes/ManiGuard fruits Jar repeat first smoothing gain.json'
with output.open('x') as stream:stream.write(json.dumps(result,indent=2)+'\n')
print(json.dumps(result['smoothing_identity_input_comparison'],indent=2))
print('All four case configurations, source snapshots and trace hashes validated.')
