def need(ok, message):
    if not ok:
        raise ValueError(message)
    CHECKS[message] += 1


def read(path):
    return json.loads(Path(path).read_text())


def digest(path):
    path = Path(path)
    need(path.is_file() and not path.is_symlink(), "regular immutable file")
    before = path.stat()
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            need(time.monotonic() < DEADLINE, "bounded observation deadline")
            h.update(chunk)
    after = path.stat()
    fields = ("st_dev", "st_ino", "st_size", "st_mtime_ns", "st_ctime_ns")
    need(all(getattr(before, k) == getattr(after, k) for k in fields), "file stable while hashing")
    return h.hexdigest()


def tree_hash(path):
    path = Path(path)
    need(not path.is_symlink(), "tree is not a symlink")
    if path.is_file():
        return {"kind": "file", "sha256": digest(path)}
    need(path.is_dir(), "recorded output directory exists")
    paths = sorted(path.rglob("*"))
    need(all(not p.is_symlink() for p in paths), "output tree has no symlinks")
    files = {str(p.relative_to(path)): digest(p) for p in paths if p.is_file()}
    need(paths == sorted(path.rglob("*")), "output membership stable")
    return {
        "kind": "directory",
        "sha256": hashlib.sha256(json.dumps(files, sort_keys=True).encode()).hexdigest(),
        "n_files": len(files),
    }


def same(a, b):
    if isinstance(a, dict) and isinstance(b, dict):
        return set(a) == set(b) and all(same(a[k], b[k]) for k in a)
    if isinstance(a, list) and isinstance(b, list):
        return len(a) == len(b) and all(same(x, y) for x, y in zip(a, b, strict=True))
    if isinstance(a, (int, float)) and isinstance(b, (int, float)):
        if math.isnan(a) or math.isnan(b):
            return math.isnan(a) and math.isnan(b)
        return (
            a == b if not math.isfinite(a) or not math.isfinite(b) else math.isclose(a, b, rel_tol=1e-10, abs_tol=1e-12)
        )
    return a == b


def subset(expected, saved, label):
    need(all(k in saved and same(v, saved[k]) for k, v in expected.items()), label)


def finite(x):
    return isinstance(x, (float, int)) and math.isfinite(x)


def unsafe(rule, value):
    need(finite(value), "finite saved dense clearance")
    return value <= 0 if rule == "health" else value < 0


def rates(rows, rule, source, margin):
    need(not math.isnan(margin), "decision margin is not NaN")
    n = len(rows)
    u = [unsafe(rule, r[f"cmin_dense_{rule}"]) for r in rows]
    c = [r[f"cmin_{source}_{rule}"] for r in rows]
    need(all(finite(x) for x in c), "finite saved prediction clearances")
    flags = [r.get("censored", False) for r in rows]
    need(all(type(x) is bool for x in flags), "explicit censor masks are booleans")
    a = [x >= margin for x in c]
    unresolved = [flag and not bad for flag, bad in zip(flags, u, strict=True)]
    acc = sum(a)
    fs = sum(x and y for x, y in zip(a, u, strict=True))
    uc = sum(x and y for x, y in zip(a, unresolved, strict=True))
    safe = [not x and not y for x, y in zip(u, unresolved, strict=True)]
    ns = sum(safe)
    nan = float("nan")
    return {
        "m": margin,
        "n": n,
        "n_accepted": acc,
        "acceptance_rate": acc / n if n else nan,
        "n_false_safe": fs,
        "n_censored": sum(unresolved),
        "n_accepted_censored": uc,
        "fsa_lower": fs / acc if acc else nan,
        "fsa_upper": (fs + uc) / acc if acc else nan,
        "fsa": fs / acc if acc and not uc else nan,
        "false_reject_rate": sum(not x and y for x, y in zip(a, safe, strict=True)) / ns if ns else nan,
    }


def attribution(rows, rule):
    fs = [r for r in rows if unsafe(rule, r[f"cmin_dense_{rule}"]) and r[f"cmin_imagined_{rule}"] >= 0]
    temporal = sum(r[f"cmin_endpoint_{rule}"] >= 0 for r in fs)
    readout = sum(r[f"cmin_endpoint_{rule}"] < 0 and r[f"cmin_real_readout_{rule}"] >= 0 for r in fs)
    return {
        "n_false_safe_imagined": len(fs),
        "temporal": temporal,
        "readout": readout,
        "imagination": len(fs) - temporal - readout,
    }


def gate_from_rows(rows, rules, thresholds):
    evidence = {}
    for rule in rules:
        a = attribution(rows, rule)
        fs = a["n_false_safe_imagined"]
        im = a["imagination"]
        nu = sum(unsafe(rule, r[f"cmin_dense_{rule}"]) for r in rows)
        share = im / fs if fs else None
        passes = (
            nu >= thresholds["min_unsafe"]
            and fs >= thresholds["min_false_safe"]
            and im >= thresholds["min_imagination"]
            and share is not None
            and share >= thresholds["min_share"]
        )
        evidence[rule] = {
            "n": len(rows),
            "n_unsafe": nu,
            "n_false_safe": fs,
            "n_imagination": im,
            "imagination_share": share,
            "pass": passes,
        }
    repair = [r for r in rules if evidence[r]["pass"]]
    return {
        "go": bool(repair),
        "repair_rules": repair,
        "evidence": evidence,
        "role": "development",
        "thresholds": thresholds,
        "status": "pass" if repair else "diagnostic_stop",
    }


def pair_rows(base, rows, rules):
    def ids(items):
        return [(r["bank"], r["root_id"], r["branch"]) for r in items]

    need(ids(base) == ids(rows) and len(ids(rows)) == len(set(ids(rows))), "exact unique paired row identities")
    for a, b in zip(base, rows, strict=True):
        for k in ("root", "kind", "speed_regime", "health_regime", "true_displacement"):
            need(a[k] == b[k], "paired source/regime/true-progress identity")
        need(
            type(a.get("censored", False)) is bool
            and type(b.get("censored", False)) is bool
            and a.get("censored", False) == b.get("censored", False),
            "paired censor identity",
        )
        for rule in rules:
            for source in ("dense", "endpoint"):
                for prefix in ("cmin_", "cmin_by_block_"):
                    k = f"{prefix}{source}_{rule}"
                    need(a[k] == b[k], "exact paired physical truth and horizon identity")


def paired_summary(saved, before, after, rule, m_before, m_after):
    pair_rows(before, after, [rule])
    reference, updated = rates(before, rule, "imagined", m_before), rates(after, rule, "imagined", m_after)
    subset(reference, saved["reference"], "paired reference denominators/rates")
    subset(updated, saved["updated"], "paired updated denominators/rates")
    point = updated["fsa"] - reference["fsa"]
    defined = finite(point)
    need(same(saved["point"], point) and saved["defined"] is defined, "paired point and undefined semantics")
    lo, hi = saved["lo"], saved["hi"]
    need(
        (finite(lo) and finite(hi) and -1 <= lo <= hi <= 1) or (not finite(lo) and not finite(hi)),
        "paired interval endpoints coherent",
    )
    if not defined:
        need(math.isnan(lo) and math.isnan(hi), "undefined full-data contrast has no directional interval")
    direction = (
        "decrease"
        if defined and finite(hi) and hi < 0
        else "increase"
        if defined and finite(lo) and lo > 0
        else "inconclusive"
        if defined
        else "undefined"
    )
    reason = (
        "zero_acceptance"
        if not reference["n_accepted"] or not updated["n_accepted"]
        else "accepted_censored_futures"
        if not defined
        else None
    )
    subset(
        {
            "direction": direction,
            "undefined_reason": reason,
            "interval_excludes_zero": direction in ("decrease", "increase"),
            "cluster_unit": "source_episode",
            "n_source_episodes": len({r["root"] for r in after}),
            "n_roots": len({r["root_id"] for r in after}),
        },
        saved,
        "paired direction/clusters/support semantics",
    )
    need(
        saved["n_boot_requested"] == 1000
        and type(saved["n_boot"]) is int
        and 0 <= saved["n_boot"] <= 1000
        and saved["n_boot_undefined"] == 1000 - saved["n_boot"],
        "paired usable bootstrap counts",
    )


def source_summary(saved, rows, rule, *, expanded):
    subset(
        {"n": len(rows), "n_unsafe": sum(unsafe(rule, r[f"cmin_dense_{rule}"]) for r in rows)},
        saved,
        "decomposition support counts",
    )
    if expanded:
        subset(
            {"n_source_episodes": len({r["root"] for r in rows}), "margin": 0.0}, saved, "decomposition source support"
        )
    for source in SOURCES:
        subset(rates(rows, rule, source, 0.0), saved[source], "four-source decision rates")
    if expanded:
        need(same(saved["attribution"], attribution(rows, rule)), "first-source error attribution counts")


def decomposition(saved, rows, rule):
    source_summary(saved["overall"], rows, rule, expanded=True)
    need(set(saved["by_horizon"]) == {str(i) for i in range(1, 11)}, "ten cumulative horizons recorded")

    def prefix(items, k):
        return [{**r, **{f"cmin_{s}_{rule}": r[f"cmin_by_block_{s}_{rule}"][k - 1] for s in SOURCES}} for r in items]

    for k in range(1, 11):
        need(same(saved["by_horizon"][str(k)]["seconds"], k * 0.08), "correct horizon seconds")
        source_summary(saved["by_horizon"][str(k)], prefix(rows, k), rule, expanded=True)
    for field in ("kind", "speed_regime", "health_regime"):
        need(set(saved["by_regime"][field]) == {r[field] for r in rows}, "all recorded regimes retained")
        for value, block in saved["by_regime"][field].items():
            items = [r for r in rows if r[field] == value]
            source_summary(block["overall"], items, rule, expanded=True)
            for k in range(1, 11):
                source_summary(block["by_horizon"][str(k)], prefix(items, k), rule, expanded=True)

def terminal(state, stage_id, paths):
    stage = state["stages"][stage_id]
    need(stage["status"] in ("completed", "gate_stopped"), "requested stage is terminal")
    attempts = stage.get("attempts", [])
    need(
        attempts and attempts[-1].get("finished_unix") is not None and attempts[-1].get("returncode") in (0, 2),
        "terminal subprocess has a recorded finish",
    )
    for path in paths:
        need(
            str(path) in stage["output_hashes"] and tree_hash(path) == stage["output_hashes"][str(path)],
            "frozen terminal output tree identity",
        )
    return stage


def reference(path):
    for name in ("hf_upload.json", "upload_receipt.json"):
        if (path / name).is_file():
            r = read(path / name)
            return {k: r[k] for k in ("repo_id", "revision", "path") if k in r}
    raise ValueError("Missing recorded upstream upload receipt")


def checked_export(ref, expected_name):
    need(ref["file"] == expected_name, "expected saved row filename")
    for folder in (RUN, RESULTS, BANK):
        p = folder / expected_name
        need(digest(p) == ref["sha256"] and p.stat().st_size == ref["bytes"], "mirrored saved row hash and byte size")
    return read(RUN / expected_name)


def bank_rows(name, rows, cfg, rules, source_id, split):
    n_roots = cfg["dev_roots"] if name == "dev" else cfg["test_roots"]
    meta = read(BANK / name / "roots.json")
    roots = meta["roots"]
    need(
        len(roots) == len({r["root_id"] for r in roots}) == len({r["episode"] for r in roots}) == n_roots,
        "declared distinct evaluation roots",
    )
    need(meta["n_branches"] == len(rows) == n_roots * cfg["tapes"], "declared evaluation branch count")
    role = "development" if name == "dev" else "test"
    need(all(r["episode"] in split["roots"][role] for r in roots), "evaluation root source roles")
    expected = {
        "source": source_id,
        "seed": 20261020,
        "kind": "stress" if name == "stress" else "representative",
        "role": role,
        "n_roots": n_roots,
        "n_tapes": cfg["tapes"],
        "frameskip": 10,
        "history": 3,
        "horizon": 10,
        "schema": 2,
    }
    need(
        meta["meta"]["provenance"] == expected and meta["meta"]["charged_steps_per_branch"] == 100,
        "frozen evaluation bank provenance",
    )
    by_id = {r["root_id"]: r for r in roots}
    need(
        Counter(r["root_id"] for r in rows) == Counter({r["root_id"]: cfg["tapes"] for r in roots}),
        "every declared evaluation root has all tapes",
    )
    need(sorted(r["branch"] for r in rows) == list(range(len(rows))), "unique complete branch index range")
    for row in rows:
        need(
            row["bank"] == name and row["root"] == by_id[row["root_id"]]["episode"],
            "row source identity matches bank metadata",
        )
        for rule in rules:
            for source in SOURCES:
                key = f"cmin_by_block_{source}_{rule}"
                if key not in row:
                    continue
                vals = row[key]
                need(
                    len(vals) == 10
                    and all(finite(v) for v in vals)
                    and all(b <= a for a, b in zip(vals, vals[1:]))
                    and same(vals[-1], row[f"cmin_{source}_{rule}"]),
                    "complete cumulative horizon clearance identity",
                )


def evaluation(saved, rows_by_bank, baseline, rules, baseline_margins):
    for name, rows in rows_by_bank.items():
        pair_rows(baseline[name], rows, rules)
        computed = {
            "state_mae_by_block": np.mean([r["state_abs_error_by_block"] for r in rows], axis=0).tolist(),
            "displacement_mae": float(
                np.mean([abs(r["predicted_displacement"] - r["true_displacement"]) for r in rows])
            ),
        }
        subset(computed, saved[name], "saved task-progress means")
        ordinary = [r for r in rows if r["kind"] == "policy"]
        subset(
            {
                "n": len(ordinary),
                "state_mae_by_block": np.mean([r["state_abs_error_by_block"] for r in ordinary], axis=0).tolist()
                if ordinary
                else None,
                "displacement_mae": float(
                    np.mean([abs(r["predicted_displacement"] - r["true_displacement"]) for r in ordinary])
                )
                if ordinary
                else None,
            },
            saved[name]["ordinary_retention"],
            "ordinary-policy retention means and support",
        )
        for rule in rules:
            metric = saved[name][rule]
            margin = saved["dev"][rule]["margin"]
            need(same(metric["margin"], margin), "final-test margin equals recorded development margin")
            subset(rates(rows, rule, "imagined", margin), metric["at_matched"], "matched-margin rates and denominators")
            paired_summary(metric["paired_fsa_difference"], baseline[name], rows, rule, baseline_margins[rule], margin)


def executed_round(path, expected_ids, candidates, acq_roots):

    need(
        meta["roots"] == [acq_roots[i] for i in ordered_roots],
        "selected root metadata matches frozen acquisition roots",
    )

                subset(costs, point, "charged simulator/model/training budgets")
                need(
                    finite(point["training_time_s"]) and point["training_time_s"] >= 0, "recorded finite training time"
                )
                executed_round(BANK / f"{arm}_s{seed}_b{budget}", selected[previous:], candidates, acq_roots)
                checkpoint = RUN / f"{arm}-s{seed}-b{budget}"
                sel = read(checkpoint / "selection.json")
                pm = read(checkpoint / "manifest.json")
                check_checkpoint_provenance(pm, seed, manifests[0]["upstream_revisions"])
                need(
                    sel["selected_ids"] == selected
                    and sel["seed_ids"] == sorted(seed_ids)
                    and (ROOT / sel["candidate_file"]).resolve() == pool_path
                    and sel["candidate_sha256"] == digest(pool_path),
                    "checkpoint-to-candidate identity",
                )
                subset(costs, pm["costs"], "checkpoint manifest cost identity")
                pcfg = yaml.safe_load((checkpoint / "config.yaml").read_text())
                check_adaptation_config(pcfg["adaptation"], report["adaptation_recipe"], seed)
                need(
                    {k: v for k, v in pcfg.items() if k != "adaptation"} == cfg,
                    "checkpoint exact saved training configuration",
                )
                need(
                    digest(checkpoint / "scalers.npz") == digest(MODEL / "scalers.npz"),
                    "unchanged adapted-checkpoint action scalers",
                )
                need(
                    pm["data"]["model_sha256"] == s3["model_sha256"]
                    and pm["data"]["probe_sha256"] == s3["probe_files_sha256"]["walker_mlp.pt"]
                    and pm["data"]["source"] == source_id
                    and pm["data"]["gate_sha256"] == digest(PROBES / "gate.json")
                    and pm["data"]["weights_sha256"] == digest(checkpoint / "weights.pt"),
                    "adapted model/probe/source/gate/weights identity",
                )
                p = checkpoint / "evaluation_rows.json"
                ref = point["evaluation_rows"]
                need(
                    ref["file"] == str(p.relative_to(RUN)) and ref["sha256"] == digest(p),
                    "adapted row export hash identity",
                )
                rows = read(p)
                need(set(rows) == set(baseline), "adapted exports retain all banks")
                evaluation(point["eval"], rows, baseline, rules, base_margins)
                need(same(pm["metrics"], point["eval"]), "checkpoint metric export equality")
                decomposed = read(checkpoint / "decomposition.json")
                for name, items in rows.items():
                    merged = [
                        {
                            **r,
                            **{
                                k: v
                                for k, v in b.items()
                                if k.startswith(("cmin_real_readout_", "cmin_by_block_real_readout_"))
                            },
                        }
                        for r, b in zip(items, baseline[name], strict=True)
                    ]
                    for rule in rules:
                        decomposition(decomposed[rule][name], merged, rule)
                receipt = read(checkpoint / "hf_upload.json")
                need(
                    receipt["repo_type"] == "model"
                    and receipt["repo_id"] == reference(MODEL)["repo_id"]
                    and receipt["run_id"] == pm["run_id"] == f"{RUN_ID}-{arm}-s{seed}-b{budget}"
                    and receipt["path"] == "adapted/" + receipt["run_id"]
                    and receipt["revision"] == point["hf_revision"]
                    and bool(re.fullmatch("[0-9a-f]{40}", receipt["revision"])),
                    "pinned adapted-model runtime upload receipt",
                )
                receipts.append(receipt)
                saved_rows[arm, seed, budget] = rows
                points[arm, seed, budget] = point
                previous = budget
    need(len(receipts) == 12, "all twelve adapted-model receipts present")
    paired = report["boundary_vs_random"]
    subset(
        {"contrast": "boundary minus random", "cluster_unit": "source_episode", "confidence_level": 0.95},
        paired,
        "declared paired contrast semantics",
    )
    need(set(paired["by_budget"]) == {"128", "512"}, "paired summary has every budget")
    for budget in BUDGETS:
        need(set(paired["by_budget"][str(budget)]["by_seed"]) == {"0", "1", "2"}, "paired summary has every seed")
        for seed in SEEDS:
            item = paired["by_budget"][str(budget)]["by_seed"][str(seed)]
            need(
                item["charged_steps"]
                == points["random", seed, budget]["charged_steps"]
                == points["boundary", seed, budget]["charged_steps"],
                "paired comparison exactly charged",
            )
            for arm in ARMS:
                need(
                    item["evaluation_rows_sha256"][arm] == points[arm, seed, budget]["evaluation_rows"]["sha256"],
                    "paired summary row hash identity",
                )
            for name in ("test", "stress"):
                for rule in rules:
                    paired_summary(
                        item[name][rule],
                        saved_rows["random", seed, budget][name],
                        saved_rows["boundary", seed, budget][name],
                        rule,
                        points["random", seed, budget]["eval"]["dev"][rule]["margin"],
                        points["boundary", seed, budget]["eval"]["dev"][rule]["margin"],
                    )
    need(report["evaluation_steps"] == (24 + 64 + 64) * 8 * 100, "separate complete evaluation charge")
    return baseline["dev"], receipts
