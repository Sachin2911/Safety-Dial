def sha(b):return hashlib.sha256(b).hexdigest()
def statid(p):
 s=p.stat();return [s.st_dev,s.st_ino,s.st_size,s.st_mtime_ns,s.st_ctime_ns]
def stable_read(p):
 assert p.is_file() and not p.is_symlink(),'Expected regular nonsymlink source'
 a=statid(p);b=p.read_bytes();assert statid(p)==a,'Source changed during read';return b,a

 selected=[]

 (OUT/'README.md').write_text(readme)
 for b,row in selected:assert stable_read(ROOT/row['source_path'])==(b,row['stat_identity'])
