def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(path):
    path = Path(path)
    require(path.is_file() and not path.is_symlink(), f'Not a regular file: {path}')
    before = path.stat()
    h = hashlib.sha256()
    with path.open('rb') as handle:
        for chunk in iter(lambda: handle.read(1024 * 1024), b''):
            h.update(chunk)
    after = path.stat()
    require((before.st_size, before.st_mtime_ns) == (after.st_size, after.st_mtime_ns),
            f'File changed during audit: {path}')
    return h.hexdigest()


def read_json(path):
    return json.loads(Path(path).read_text())


def safe(value):
    if isinstance(value, dict):
        return {str(k): safe(v) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [safe(v) for v in value]
    if isinstance(value, np.ndarray):
        return safe(value.tolist())
    if isinstance(value, np.generic):
        return safe(value.item())
    if isinstance(value, float) and not math.isfinite(value):
        return None
    return value


def equal(actual, expected, label):
    require(safe(actual) == safe(expected), f'Inconsistent {label}')


def receipt(directory):
    for name in ('hf_upload.json', 'upload_receipt.json'):
        path = Path(directory) / name
        if path.is_file():
            result = read_json(path)
            require(re.fullmatch(r'[0-9a-f]{40}', result.get('revision', '')) is not None,
                    f'Unpinned receipt: {path}')
            require(bool(result.get('repo_id')) and bool(result.get('path')), f'Bad receipt: {path}')
            return {k: result[k] for k in ('repo_id', 'revision', 'path')}
    raise ValueError(f'Missing local runtime receipt: {directory}')


def barrier(mode):
    """Read only workflow state until the requested stages are terminal."""
    cfg = yaml.safe_load(CONFIG.read_text())
    equal(cfg['workflow_id'], WORKFLOW_ID, 'recovery workflow identity')
    equal((REPO / cfg['state_dir']).resolve(), (REPO / 'runs/workflows' / WORKFLOW_ID).resolve(),
          'recovery workflow state directory')
    recipes = {stage['id']: stage for stage in cfg['stages']}
    def argument(stage, option):
        argv = recipes[stage]['argv']
        require(argv.count(option) == 1, f'Expected exactly one {stage} {option}')
        return argv[argv.index(option) + 1]
    for stage, run_dir in (('lewm_a_full', RUN), ('physical_probe_gate', S3_RUN),
                           ('prospective_acquisition', S4_RUN)):
        equal(argument(stage, '--run-id'), run_dir.name, f'{stage} fresh run ID')
    equal((REPO / argument('lewm_a_full', '--resume-from')).resolve(), RESUME_BUNDLE.resolve(),
          'workflow resume bundle path')
    for stage in ('physical_probe_gate', 'prospective_acquisition'):
        equal((REPO / argument(stage, '--model')).resolve(), RUN.resolve(), f'{stage} recovered model')
        equal((REPO / argument(stage, '--data-dir')).resolve(), DATA.resolve(), f'{stage} original data')
    equal((REPO / argument('prospective_acquisition', '--probes')).resolve(), S3_RUN.resolve(),
          'S4 recovered probe run')
    state_path = REPO / cfg['state_dir'] / 'state.json'
    state = read_json(state_path)
    require(state['config_sha256'] == digest(CONFIG), 'Workflow configuration hash changed')
    requested = ['lewm_a_full'] + (['physical_probe_gate'] if mode == 'both' else [])
    stages = {}
    for name in requested:
        stage = state['stages'][name]
        allowed = ['completed'] if name == 'lewm_a_full' else ['completed', 'gate_stopped']
        require(stage['status'] in allowed, f'Wait for finalized {name}; no artifact reads authorized')
        attempts = stage.get('attempts', [])
        require(bool(attempts) and attempts[-1].get('finished_unix') is not None,
                f'No completed subprocess for {name}')
        require(attempts[-1].get('returncode') in ([0] if name == 'lewm_a_full' else [0, 2]),
                f'Unsuccessful subprocess for {name}')
        require(bool(stage.get('output_hashes')), f'Missing output hashes for {name}')
        stages[name] = stage
    sys.path.insert(0, str(REPO / 'experiments'))
    from helpers.managedWorkflow import output_hash
    for name, stage in stages.items():
        for path, expected in stage['output_hashes'].items():
            equal(output_hash(Path(path)), expected, f'{name} output {path}')
    return stages, cfg

def fsa_from_rows(clearance, unsafe):
    accepted = clearance >= 0
    n, na, nu, fs = len(unsafe), int(accepted.sum()), int(unsafe.sum()), int((accepted & unsafe).sum())
    value = fs / na if na else None
    return {'m': 0.0, 'n': n, 'n_accepted': na, 'acceptance_rate': na / n,
            'n_false_safe': fs, 'n_censored': 0, 'n_accepted_censored': 0,
            'fsa_lower': value, 'fsa_upper': value, 'fsa': value,
            'false_reject_rate': int((~accepted & ~unsafe).sum()) / (n - nu) if n > nu else None}


def audit_s3(stage, source_bytes):
    import h5py
    from helpers.walkerValidation import gate_decision, load_source_split
    report, rows = read_json(S3_RUN / 'gate.json'), read_json(S3_RUN / 'gate_rows.json')
    cfg = yaml.safe_load((S3_RUN / 'config.yaml').read_text())

    equal(report['run_id'], S3_RUN.name, 'S3 report run ID')
    equal(manifest['run_id'], S3_RUN.name, 'S3 manifest run ID')
    equal(report['role'], 'development', 'S3 qualification role')
    equal((REPO / cfg['model']).resolve(), RUN.resolve(), 'S3 selected model')
    equal((REPO / cfg['data_dir']).resolve(), DATA.resolve(), 'S3 selected source data')
    equal((cfg['n_roots'], cfg['seed'], cfg['min_unsafe'], cfg['probe_frames']), (64, 0, 10, 30000), 'S3 recipe')
    equal(report['splits_sha256'], digest(DATA / 'splits.json'), 'frozen S1 source split')
    roles = load_source_split(DATA)
    for name, expected in report['model_files_sha256'].items():
        equal(digest(RUN / name), expected, f'S3 model file {name}')
    equal(report['model_sha256'], digest(RUN / 'weights.pt'), 'S3 model weight identity')
    equal(manifest['data']['model_sha256'], report['model_sha256'], 'S3 manifest weight identity')
    for name, expected in report['probe_files_sha256'].items():
        equal(digest(S3_RUN / name), expected, f'S3 probe {name}')
    equal(manifest['metrics'], report['probes'], 'S3 probe metrics')
    equal(manifest['data']['roots'], report['source_identity'], 'S3 root source identity')
    sources = {}
    for key, name in (('roots', 'roots.h5'), ('probe', 'probe.h5')):
        identity, path = manifest['data'][key], DATA / name
        equal(Path(identity['path']).resolve(), path.resolve(), f'{key} source path')
        equal(identity['size'], path.stat().st_size, f'{key} source size')
        if source_bytes:
            equal(digest(path), identity['sha256'], f'{key} source bytes')
        with h5py.File(path, 'r') as source:
            sources[key] = np.asarray(source['ep_len'][:], dtype=int)
    # Reproduce only RNG source selection from lengths, not frames, actions or outcomes.
    rng = np.random.default_rng(cfg['seed'])
    frames, selected_probe = 0, []
    for episode, n in enumerate(sources['probe']):
        selected_probe.append(episode)
        frames += (int(n) + 1) // 2
        if frames >= cfg['probe_frames']:
            break
    val = sorted(rng.choice(selected_probe, size=max(1, int(len(selected_probe) * .2)), replace=False).tolist())
    train = sorted(set(selected_probe) - set(val))
    equal(manifest['data']['probe_train_episodes'], train, 'S3 deterministic probe train episodes')
    equal(manifest['data']['probe_val_episodes'], val, 'S3 deterministic probe validation episodes')
    equal(manifest['data']['n_probe_frames'], frames, 'S3 probe frame count')
    expected_roots = []
    for episode, n in enumerate(sources['roots']):
        if episode not in roles['roots']['development'] or n < 121:
            continue
        positions = rng.choice(np.arange(20, n - 100), size=min(2, int(n) // 200 + 1), replace=False)
        for position in positions:
            expected_roots.append(f'w-e{episode}-t{position}')
            for sigma in (.1, .3):
                rng.normal(0, sigma, (100, 6))  # Consume declared tape-draw RNG, no controls queried.
        if len(expected_roots) >= cfg['n_roots']:
            break
    equal([(r['root'], r['tape']) for r in rows], [(root, tape) for root in expected_roots for tape in range(3)],
          'S3 frozen source/RNG order and exact three paired tapes')
    require(len(expected_roots) >= cfg['n_roots'], 'Insufficient S3 source roots')
    source_ids = sorted({int(re.fullmatch(r'w-e(\d+)-t(\d+)', root).group(1)) for root in expected_roots})
    equal(manifest['data']['gate_source_episodes'], source_ids, 'S3 gate source episode identity')
    require(not set(source_ids) & (set(roles['roots']['test']) | set(roles['roots']['acquisition'])),
            'S3 heldout/acquisition source leakage')
    require(all(math.isfinite(r[f'{source}_{rule}']) for r in rows
                for source in ('true', 'real', 'imag') for rule in ('health', 'speed')),
            'Nonfinite saved paired S3 clearances')
    diagnostics, decisions = {}, {}
    for rule in ('health', 'speed'):
        truth = np.asarray([r[f'true_{rule}'] for r in rows])
        unsafe = truth <= 0 if rule == 'health' else truth < 0
        real = np.asarray([r[f'real_{rule}'] for r in rows])
        imag = np.asarray([r[f'imag_{rule}'] for r in rows])
        nu, ns = int(unsafe.sum()), int((~unsafe).sum())
        tp, tn = int(((real < 0) & unsafe).sum()), int(((real >= 0) & ~unsafe).sum())
        diagnostics[rule] = {'n_unsafe': nu, 'n_safe': ns, 'unsafe_detected': tp,
                             'safe_accepted': tn, 'unsafe_recall': tp / nu if nu else None,
                             'specificity': tn / ns if ns else None}
        decisions[rule] = {'n': len(rows), 'n_unsafe': nu, 'real_readout': fsa_from_rows(real, unsafe),
                           'imagined': fsa_from_rows(imag, unsafe),
                           'real_agreement': float(np.mean((real < 0) == unsafe))}
    equal(report['decision_diagnostics'], diagnostics, 'S3 row-derived decision diagnostics')
    equal(report['decisions_m0'], decisions, 'S3 paired row-derived FSA and counts')
